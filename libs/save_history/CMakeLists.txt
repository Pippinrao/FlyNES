cmake_minimum_required(VERSION 3.16)
project(save_history VERSION 1.0.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(SAVE_HISTORY_BUILD_TESTS "Build standalone tests" OFF)
add_library(save_history STATIC src/save_history.cpp vendor/sqlite/sqlite3.c)
target_include_directories(save_history PRIVATE vendor/sqlite)
target_compile_definitions(save_history PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
if(UNIX)
  find_package(Threads REQUIRED)
  target_link_libraries(save_history PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()
add_library(save_history::save_history ALIAS save_history)
target_compile_features(save_history PUBLIC cxx_std_17)
set_target_properties(save_history PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(save_history PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
install(TARGETS save_history EXPORT save_historyTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/save_history)
install(FILES vendor/sqlite/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/save_history RENAME SQLITE-NOTICE.md)
install(EXPORT save_historyTargets NAMESPACE save_history:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/save_history)
configure_package_config_file(cmake/save_historyConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/save_historyConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/save_history)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/save_historyConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/save_historyConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/save_historyConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/save_history)
if(SAVE_HISTORY_BUILD_TESTS)
  enable_testing()
  add_executable(save_history_test tests/save_history_test.cpp)
  target_link_libraries(save_history_test PRIVATE save_history)
  target_include_directories(save_history_test PRIVATE vendor/sqlite)
  foreach(case open storage retention quota integrity restore restore_failure commit_failure migration recovery)
    add_test(NAME save_history_${case} COMMAND save_history_test ${case})
  endforeach()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME save_history_killed_transaction COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/killed_transaction.py $<TARGET_FILE:save_history_test>)
endif()
