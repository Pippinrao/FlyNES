cmake_minimum_required(VERSION 3.16)
project(save_history_external_consumers LANGUAGES C CXX)
find_package(save_history 1 CONFIG REQUIRED)
enable_testing()
add_executable(save_history_c_consumer consumer.c)
target_link_libraries(save_history_c_consumer PRIVATE save_history::save_history)
add_test(NAME c_consumer COMMAND save_history_c_consumer)
add_executable(save_history_cpp_consumer consumer.cpp)
target_link_libraries(save_history_cpp_consumer PRIVATE save_history::save_history)
add_test(NAME cpp_consumer COMMAND save_history_cpp_consumer)
